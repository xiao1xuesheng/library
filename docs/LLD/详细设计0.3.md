## 2. 数据库表设计

### 2.1 真实客户表（t_user）

```sql
CREATE TABLE t_user (
    id                  BIGINT AUTO_INCREMENT PRIMARY KEY,
    username            VARCHAR(64)  NOT NULL UNIQUE COMMENT '用户名',
    password_hash       VARCHAR(128) NOT NULL COMMENT '密码哈希',
    email               VARCHAR(128) NULL UNIQUE COMMENT '邮箱',
    phone               VARCHAR(32)  NULL UNIQUE COMMENT '手机号',
    user_type           VARCHAR(32)  NOT NULL DEFAULT 'NORMAL' COMMENT '用户类型：NORMAL / VIP',
    user_status         VARCHAR(32)  NOT NULL DEFAULT 'NORMAL' COMMENT '用户状态：NORMAL / MUTED / DISABLED',
    vip_expire_at       TIMESTAMP    NULL COMMENT 'VIP到期时间',
    auto_renew          TINYINT(1)   NOT NULL DEFAULT 0 COMMENT '是否开通自动续期：0-否 1-是',
    alert_preference    VARCHAR(32)  NOT NULL DEFAULT 'REALTIME' COMMENT '告警偏好：REALTIME / DAILY_SUMMARY / DO_NOT_DISTURB',
    created_at          TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_user_type (user_type),
    INDEX idx_user_status (user_status),
    INDEX idx_vip_expire_at (vip_expire_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='真实客户表';
```

**说明**：
- `user_type`：NORMAL（普通用户）、VIP（VIP用户）
- `user_status`：NORMAL（正常）、MUTED（禁言）、DISABLED（禁用）
- `auto_renew`：VIP自动续期开关
- `alert_preference`：告警接收偏好
- 借阅限制和期限不再存储在用户表中，而是通过Nacos配置中心动态获取

### 2.2 管理员表（t_admin，新增）

```sql
CREATE TABLE t_admin (
    id                  BIGINT AUTO_INCREMENT PRIMARY KEY,
    username            VARCHAR(64)  NOT NULL UNIQUE COMMENT '管理员用户名',
    password_hash       VARCHAR(128) NOT NULL COMMENT '密码哈希',
    role                VARCHAR(32)  NOT NULL DEFAULT 'ADMIN' COMMENT '角色：SUPER_ADMIN / ADMIN',
    status              VARCHAR(32)  NOT NULL DEFAULT 'ACTIVE' COMMENT '状态：ACTIVE / DISABLED',
    created_by          BIGINT       NULL COMMENT '创建人ID',
    created_at          TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_role (role),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='管理员表';
```

**说明**：
- 管理员与真实客户完全分离，使用独立的表
- `role`：SUPER_ADMIN（超级管理员，可创建管理员）、ADMIN（普通管理员）
- 管理员由BOS后台创建和授权，不走注册流程

```java
@Service
public class BookCacheManager {
    
    private final RedisTemplate<String, Object> redisTemplate;
    private final CaffeineCache localCache;
    
    // 获取书籍详情（二级缓存）
    public Book getBookDetail(Long bookId) {
        // 1. 先查本地缓存
        Book book = localCache.getIfPresent("book:" + bookId);
        if (book != null) {
            return book;
        }
        
        // 2. 查Redis缓存
        String key = "elibrary:book:info:" + bookId;
        book = (Book) redisTemplate.opsForHash().entries(key);
        if (book != null) {
            localCache.put("book:" + bookId, book);
            return book;
        }
        
        // 3. 查数据库
        book = bookRepository.findById(bookId).orElse(null);
        if (book != null) {
            redisTemplate.opsForHash().putAll(key, book.toMap());
            redisTemplate.expire(key, 30, TimeUnit.MINUTES);
            localCache.put("book:" + bookId, book);
        }
        
        return book;
    }
    
    // 更新书籍信息时删除缓存
    public void evictBookCache(Long bookId) {
        redisTemplate.delete("elibrary:book:info:" + bookId);
        localCache.invalidate("book:" + bookId);
        // 延迟双删
        CompletableFuture.runAsync(() -> {
            try {
                Thread.sleep(1000);
                redisTemplate.delete("elibrary:book:info:" + bookId);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        });
    }
}
```

```java
@Service
public class AlertFingerprintCacheManager {
    
    private final RedisTemplate<String, Object> redisTemplate;
    
    // 检查告警是否在静默窗口内
    public boolean isInSilenceWindow(String fingerprint) {
        String key = "elibrary:alert:fingerprint:" + fingerprint;
        return Boolean.TRUE.equals(redisTemplate.hasKey(key));
    }
    
    // 记录告警指纹
    public void recordAlertFingerprint(String fingerprint, AlertRecord record) {
        String key = "elibrary:alert:fingerprint:" + fingerprint;
        
        Map<String, Object> data = new HashMap<>();
        data.put("count", record.getTriggerCount());
        data.put("firstTime", record.getFirstTriggerTime().toString());
        data.put("lastTime", record.getLastTriggerTime().toString());
        data.put("status", record.getStatus());
        
        redisTemplate.opsForHash().putAll(key, data);
        // TTL根据告警类型设置
        redisTemplate.expire(key, getSilenceTTL(record.getAlertType()), TimeUnit.HOURS);
    }
    
    // 获取告警触发次数
    public int getTriggerCount(String fingerprint) {
        String key = "elibrary:alert:fingerprint:" + fingerprint;
        Object count = redisTemplate.opsForHash().get(key, "count");
        return count != null ? Integer.parseInt(count.toString()) : 0;
    }
    
    // 根据告警类型获取TTL
    private long getSilenceTTL(String alertType) {
        switch (alertType) {
            case "BALANCE_LOW":
            case "FORCE_RETURN":
                return 24; // 24小时
            case "VIP_EXPIRY":
                return 168; // 7天
            case "PAYMENT_FAILED":
                return 1; // 1小时
            default:
                return 24;
        }
    }
}
```

| 借阅记录创建 | 更新缓存 | 创建后更新借阅数缓存 |

```java
@Component
public class CacheBreakdownProtector {
    
    private final RedissonClient redissonClient;
    
    public Object getWithLock(String key, Supplier<Object> loader) {
        String lockKey = "cache:lock:" + key;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            if (lock.tryLock(3, 10, TimeUnit.SECONDS)) {
                try {
                    // 双重检查
                    Object cached = redisTemplate.opsForValue().get(key);
                    if (cached != null) {
                        return cached;
                    }
                    return loader.get();
                } finally {
                    lock.unlock();
                }
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
        
        return null;
    }
}
```

```java
@Component
public class BookParseEventConsumer {
    
    private final FormatConversionService formatConversionService;
    private final BookRepository bookRepository;
    
    @KafkaListener(
        topics = "elibrary-book-parse-events",
        groupId = "book-parse-service",
        concurrency = "3"
    )
    public void consume(String message) {
        BookParseEvent event = JsonUtil.fromJson(message, BookParseEvent.class);
        
        // 幂等性检查
        if (isDuplicate(event.getEventId())) {
            return;
        }
        
        try {
            // 根据文件格式选择解析器
            switch (event.getFileFormat()) {
                case "pdf":
                    formatConversionService.convertPdfToEpub(event);
                    break;
                case "doc":
                case "docx":
                    formatConversionService.convertWordToEpub(event);
                    break;
                case "txt":
                    formatConversionService.convertTxtToEpub(event);
                    break;
                default:
                    throw new BusinessException(ErrorCode.UNSUPPORTED_FILE_FORMAT);
            }
            
            // 更新书籍状态
            bookRepository.updateStatus(event.getBookId(), "ONLINE");
            
            markAsConsumed(event.getEventId());
        } catch (Exception e) {
            log.error("Failed to parse book: bookId={}", event.getBookId(), e);
            // 重试3次后标记为解析失败
            handleParseFailure(event, e);
        }
    }
}
```

## 5. 分布式锁设计

### 5.1 Redis分布式锁实现

```java
@Component
public class DistributedLockManager {
    
    private final RedissonClient redissonClient;
    
    // 获取锁
    public boolean tryLock(String lockKey, long waitTime, long leaseTime) {
        RLock lock = redissonClient.getLock(lockKey);
        try {
            return lock.tryLock(waitTime, leaseTime, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return false;
        }
    }
    
    // 释放锁
    public void unlock(String lockKey) {
        RLock lock = redissonClient.getLock(lockKey);
        if (lock.isHeldByCurrentThread()) {
            lock.unlock();
        }
    }
    
    // 带WatchDog的锁（自动续期）
    public boolean tryLockWithWatchDog(String lockKey) {
        RLock lock = redissonClient.getLock(lockKey);
        try {
            return lock.tryLock(3, TimeUnit.SECONDS);
            // 不设置leaseTime，WatchDog会自动续期
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return false;
        }
    }
}
```

### 5.2 锁使用场景

| 场景 | 锁Key | 等待时间 | 租期 | WatchDog |
|------|-------|----------|------|----------|
| 书籍状态更新 | lock:book:status:{bookId} | 3s | 5s | 否 |
| 借阅创建 | lock:borrow:{userId}:{bookId} | 3s | 10s | 否 |
| 余额扣减 | lock:balance:{userId} | 3s | 5s | 否 |
| 支付回调处理 | lock:pay:callback:{orderNo} | 3s | 5s | 否 |
| 定时任务执行 | lock:job:{jobName} | 0s | 1h | 是 |
| 对账任务 | lock:reconciliation:{date} | 0s | 30m | 是 |
| 强制归还 | lock:force-return:{userId} | 3s | 10s | 否 |
| **书籍解析** | **lock:book-parse:{bookId}** | **3s** | **30m** | **是** |

## 7. 核心业务流程

### 7.1 借阅流程

```
用户发起借阅请求
       │
       ▼
1. 获取分布式锁 lock:borrow:{userId}:{bookId}
       │
       ▼
2. 校验用户状态（NORMAL）
       │
       ▼
3. 校验书籍状态（ONLINE）
       │
       ▼
4. 校验借阅权限（书籍访问类型）
       │  VIP免费 → 仅VIP可借阅
       │  VIP付费 → 仅VIP可借阅，按日扣费
       │  公开免费 → 所有用户可借阅
       │  公开付费 → 所有用户可借阅，按日扣费
       │
       ▼
5. 校验借阅数量（从Nacos获取配置）
       │  NORMAL用户：maxLimitFree
       │  VIP用户：maxLimitVip
       │
       ▼
6. 检查是否已有借阅记录（幂等性）
       │  已借阅 → 返回成功
       │  未借阅 → 继续
       │
       ▼
7. 计算借阅期限（从Nacos获取配置）
       │  NORMAL用户：durationFree
       │  VIP用户：durationVip
       │
       ▼
8. 如需付费，检查账户余额
       │  余额不足 → 返回错误
       │  余额充足 → 扣减首日费用
       │
       ▼
9. 创建借阅记录（事务）
       │  插入借阅记录
       │  更新书籍借阅计数
       │  记录余额变动日志（如扣费）
       │
       ▼
10. 发送借阅事件到Kafka
       │
       ▼
11. 释放分布式锁
       │
       ▼
12. 返回借阅成功
```

## 8. 鉴权与MDC实现

### 8.1 鉴权设计说明

**"统一鉴权"的含义**：
- 统一JWT解析机制（同一个Filter）
- 统一MDC注入机制（同一个上下文传递）
- 统一Token格式（同一套JWT工具类）

**数据源分离**：
- Web接口（`/api/**`）：从 `t_user` 表查询真实客户
- BOS接口（`/bos/**`）：从 `t_admin` 表查询管理员

**Token生成流程**：
```
Web登录：POST /api/auth/login
  → 查询 t_user 表
  → 验证密码
  → 生成JWT（包含 userId, userType）
  
BOS登录：POST /bos/auth/login
  → 查询 t_admin 表
  → 验证密码
  → 生成JWT（包含 adminId, adminRole）
```

### 8.2 鉴权过滤器

```java
@Component
@Order(1)
public class AuthFilter implements Filter {
    
    private final UserService userService;
    private final AdminService adminService;
    
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        HttpServletRequest httpRequest = (HttpServletRequest) request;
        String requestUri = httpRequest.getRequestURI();
        
        // 设置TraceId
        String traceId = httpRequest.getHeader("X-Trace-Id");
        if (traceId == null || traceId.isEmpty()) {
            traceId = UUID.randomUUID().toString().replace("-", "");
        }
        MDC.put("traceId", traceId);
        
        // 从JWT Token中解析用户信息
        String token = httpRequest.getHeader("Authorization");
        if (token != null && token.startsWith("Bearer ")) {
            token = token.substring(7);
            Claims claims = JwtUtil.parseToken(token);
            
            if (requestUri.startsWith("/bos/")) {
                // BOS接口：管理员鉴权
                Long adminId = Long.parseLong(claims.get("adminId", String.class));
                String adminRole = claims.get("adminRole", String.class);
                
                // 校验管理员状态（从 t_admin 表查询）
                Admin admin = adminService.getAdminById(adminId);
                if (admin == null || "DISABLED".equals(admin.getStatus())) {
                    throw new SecurityException("管理员不存在或已被禁用");
                }
                
                MDC.put("adminId", String.valueOf(adminId));
                MDC.put("adminRole", adminRole);
            } else if (requestUri.startsWith("/api/")) {
                // Web接口：真实客户鉴权
                Long userId = Long.parseLong(claims.get("userId", String.class));
                String userType = claims.get("userType", String.class);
                
                // 校验用户状态（从 t_user 表查询）
                User user = userService.getUserById(userId);
                if (user == null || "DISABLED".equals(user.getUserStatus())) {
                    throw new SecurityException("用户不存在或已被禁用");
                }
                
                MDC.put("userId", String.valueOf(userId));
                MDC.put("userType", userType);
            }
        }
        
        try {
            chain.doFilter(request, response);
        } finally {
            // 请求完成后清除MDC
            MDC.clear();
        }
    }
}
```

```java
        configService = NacosFactory.createConfigService(properties);
    }
    
    public String getConfig(String dataId, String groupId, String defaultValue) {
        try {
            String config = configService.getConfig(dataId, groupId, 5000);
            return config != null ? config : defaultValue;
        } catch (NacosException e) {
            log.warn("Failed to get config from Nacos, using default: dataId={}, groupId={}", 
                dataId, groupId);
            return defaultValue;
        }
    }
    
    public int getIntConfig(String dataId, String groupId, int defaultValue) {
        String value = getConfig(dataId, groupId, String.valueOf(defaultValue));
        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException e) {
            log.warn("Failed to parse int config, using default: dataId={}, value={}", 
                dataId, value);
            return defaultValue;
        }
    }
}
```

### 13.2 S3工具类

```java
@Component
public class S3Util {
    
    private final AmazonS3 s3Client;
    
    @Value("${aws.s3.bucket.books}")
    private String booksBucket;
    
    // 上传文件
    public String uploadFile(String key, File file, String contentType) {
        ObjectMetadata metadata = new ObjectMetadata();
        metadata.setContentType(contentType);
        
        s3Client.putObject(new PutObjectRequest(booksBucket, key, file)
            .withMetadata(metadata));
        
        return s3Client.getUrl(booksBucket, key).toString();
    }
    
    // 下载文件
    public InputStream downloadFile(String key) {
        S3Object object = s3Client.getObject(booksBucket, key);
        return object.getObjectContent();
    }
    
    // 删除文件
    public void deleteFile(String key) {
        s3Client.deleteObject(booksBucket, key);
    }
    
    // 生成预签名URL（用于临时访问）
    public String generatePresignedUrl(String key, int expirationMinutes) {
        Date expiration = new Date(System.currentTimeMillis() + expirationMinutes * 60 * 1000);
        return s3Client.generatePresignedUrl(booksBucket, key, expiration, HttpMethod.GET).toString();
    }
}
```
