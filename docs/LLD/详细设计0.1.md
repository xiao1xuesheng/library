### 2.1 真实客户表（t_user）

```sql
CREATE TABLE t_user (
    id                  BIGINT AUTO_INCREMENT PRIMARY KEY,
    username            VARCHAR(64)  NOT NULL UNIQUE COMMENT '用户名',
    password_hash       VARCHAR(128) NOT NULL COMMENT '密码哈希',
    email               VARCHAR(128) NULL UNIQUE COMMENT '邮箱',
    phone               VARCHAR(32)  NULL UNIQUE COMMENT '手机号',
    user_type           VARCHAR(32)  NOT NULL DEFAULT 'FREE' COMMENT '用户类型：FREE / VIP',
    user_status         VARCHAR(32)  NOT NULL DEFAULT 'NORMAL' COMMENT '用户状态：NORMAL / MUTED / DISABLED',
    vip_expire_at       TIMESTAMP    NULL COMMENT 'VIP到期时间',
    created_at          TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_user_type (user_type),
    INDEX idx_user_status (user_status),
    INDEX idx_vip_expire_at (vip_expire_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='真实客户表';
```

**说明**：
- `user_type`：FREE（普通用户）、VIP（VIP用户）
- `user_status`：NORMAL（正常）、MUTED（禁言）、DISABLED（禁用）
- 借阅限制和期限不再存储在用户表中，而是通过Nacos配置中心动态获取

### 3.1 Key设计规范

### 3.2 缓存策略详细设计

#### 3.2.1 书籍详情缓存

```java
@Service
public class BookCacheManager {
    
    private final RedisTemplate<String, Object> redisTemplate;
    private final CaffeineCache localCache;
    
    // 获取书籍详情（二级缓存）
    public Book getBookDetail(Long bookId) {
        // 1. 先查本地缓存
        Book book = localCache.getIfPresent("book:" + bookId);
        if (book != null) {
            return book;
        }
        
        // 2. 查Redis缓存
        String key = "elibrary:book:info:" + bookId;
        book = (Book) redisTemplate.opsForHash().entries(key);
        if (book != null) {
            localCache.put("book:" + bookId, book);
            return book;
        }
        
        // 3. 查数据库
        book = bookRepository.findById(bookId).orElse(null);
        if (book != null) {
            redisTemplate.opsForHash().putAll(key, book.toMap());
            redisTemplate.expire(key, 30, TimeUnit.MINUTES);
            localCache.put("book:" + bookId, book);
        }
        
        return book;
    }
    
    // 更新书籍信息时删除缓存
    public void evictBookCache(Long bookId) {
        redisTemplate.delete("elibrary:book:info:" + bookId);
        localCache.invalidate("book:" + bookId);
        // 延迟双删
        CompletableFuture.runAsync(() -> {
            try {
                Thread.sleep(1000);
                redisTemplate.delete("elibrary:book:info:" + bookId);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        });
    }
}
```

```java
        String balanceStr = (String) redisTemplate.opsForValue().get(key);
        
        if (balanceStr != null) {
            return new BigDecimal(balanceStr);
        }
        
        // 查数据库
        Account account = accountRepository.findByUserId(userId);
        if (account != null) {
            redisTemplate.opsForValue().set(key, account.getBalance().toString(), 5, TimeUnit.MINUTES);
            return account.getBalance();
        }
        
        return BigDecimal.ZERO;
    }
    
    // 更新余额时删除缓存
    public void evictBalanceCache(Long userId) {
        redisTemplate.delete("elibrary:user:balance:" + userId);
    }
```

```java
    public Object getWithLock(String key, Supplier<Object> loader) {
        String lockKey = "cache:lock:" + key;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            if (lock.tryLock(3, 10, TimeUnit.SECONDS)) {
                try {
                    // 双重检查
                    Object cached = redisTemplate.opsForValue().get(key);
                    if (cached != null) {

                    }
                    return loader.get();
                } finally {
                    lock.unlock();

            Thread.currentThread().interrupt();
        }
        
        return null;
    }
```

## 5. 分布式锁设计

### 5.1 Redis分布式锁实现

```java
@Component
public class DistributedLockManager {
    
    private final RedissonClient redissonClient;
    
    // 获取锁
    public boolean tryLock(String lockKey, long waitTime, long leaseTime) {
        RLock lock = redissonClient.getLock(lockKey);
        try {
            return lock.tryLock(waitTime, leaseTime, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return false;
        }
    }
    
    // 释放锁
    public void unlock(String lockKey) {
        RLock lock = redissonClient.getLock(lockKey);
        if (lock.isHeldByCurrentThread()) {
            lock.unlock();
        }
    }
    
    // 带WatchDog的锁（自动续期）
    public boolean tryLockWithWatchDog(String lockKey) {
        RLock lock = redissonClient.getLock(lockKey);
        try {
            return lock.tryLock(3, TimeUnit.SECONDS);
            // 不设置leaseTime，WatchDog会自动续期
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return false;
        }
    }
}
```

```
5. 返回删除成功
```

### 7.8 阅读进度同步流程

```
用户阅读页面
       │
       ▼

       │
       ▼
read-service接收请求
       │
       ▼

       │  Key: elibrary:reading:progress:{userId}:{bookId}

       │
       ▼

       │  均未命中 → 返回第1页
       │
       ▼
```

## 8. 鉴权与MDC实现

### 8.1 鉴权过滤器

```java
@Component
@Order(1)
public class AuthFilter implements Filter {
    
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        HttpServletRequest httpRequest = (HttpServletRequest) request;
        String requestUri = httpRequest.getRequestURI();
        
        // 设置TraceId
        String traceId = httpRequest.getHeader("X-Trace-Id");
        if (traceId == null || traceId.isEmpty()) {
            traceId = UUID.randomUUID().toString().replace("-", "");
        }
        MDC.put("traceId", traceId);
        
        // 从JWT Token中解析用户信息
        String token = httpRequest.getHeader("Authorization");
        if (token != null && token.startsWith("Bearer ")) {
            token = token.substring(7);
            Claims claims = JwtUtil.parseToken(token);
            
            if (requestUri.startsWith("/bos/")) {
                // BOS接口：管理员鉴权
                Long adminId = Long.parseLong(claims.get("adminId", String.class));
                String adminRole = claims.get("adminRole", String.class);
                MDC.put("adminId", String.valueOf(adminId));
                MDC.put("adminRole", adminRole);
            } else if (requestUri.startsWith("/api/")) {
                // Web接口：真实客户鉴权
                Long userId = Long.parseLong(claims.get("userId", String.class));
                String userType = claims.get("userType", String.class);
                MDC.put("userId", String.valueOf(userId));
                MDC.put("userType", userType);
            }
        }
        
        try {
            chain.doFilter(request, response);
        } finally {
            // 请求完成后清除MDC
            MDC.clear();
        }
    }
}
```

```java
@Component
public class NacosConfigManager {
    
    private static final Logger log = LoggerFactory.getLogger(NacosConfigManager.class);
    private final ConfigService configService;
    
    @PostConstruct
    public void init() throws NacosException {
        // 初始化Nacos配置服务
        Properties properties = new Properties();
        properties.put("serverAddr", "${nacos.server-addr}");
        properties.put("namespace", "${nacos.namespace}");
        configService = NacosFactory.createConfigService(properties);
    }
    
    // 获取配置
    public String getConfig(String dataId, String groupId, String defaultValue) {
        try {
            String config = configService.getConfig(dataId, groupId, 5000);
            return config != null ? config : defaultValue;
        } catch (NacosException e) {
            log.warn("Failed to get config from Nacos, using default: dataId={}, groupId={}", 
                dataId, groupId);
            return defaultValue;
        }
    }
    
    // 获取整型配置
    public int getIntConfig(String dataId, String groupId, int defaultValue) {
        String value = getConfig(dataId, groupId, String.valueOf(defaultValue));
        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException e) {
            log.warn("Failed to parse int config, using default: dataId={}, value={}", 
                dataId, value);
            return defaultValue;
        }
    }
}
```

```java
@RestControllerAdvice
public class GlobalExceptionHandler {
    
    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<ErrorResponse> handleBusinessException(BusinessException e) {
        log.warn("Business exception: code={}, message={}", e.getCode(), e.getMessage());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
            .body(new ErrorResponse(e.getCode(), e.getMessage()));
    }
    
    @ExceptionHandler(SecurityException.class)
    public ResponseEntity<ErrorResponse> handleSecurityException(SecurityException e) {
        log.warn("Security exception: message={}", e.getMessage());
        return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
            .body(new ErrorResponse(401, e.getMessage()));
    }
    
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> handleException(Exception e) {
        log.error("Unexpected exception", e);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body(new ErrorResponse(500, "服务器内部错误"));
    }
}
```
